//! SCRATCH generator, run inside a v0.5.1 checkout only (never committed there). Builds a real
//! 0.5.1 custody vault with the v0.5.1 kit (CustodyVault + wallet::import + put), the same calls
//! the app makes, through an in-memory Keyring. Writes `custody.enc` and the keyring entries the
//! vault created (base64), plus the 0.5.1 kit config.json in the tauri-plugin-store 2.4.3 layout.
use base64::Engine;
use citrate_core_kit::custody::{CustodyError, CustodyVault, Keyring};
type Result<T> = std::result::Result<T, CustodyError>;
use std::collections::BTreeMap;
use std::sync::{Arc, Mutex};

#[derive(Clone, Default)]
struct MapKeyring(Arc<Mutex<BTreeMap<String, Vec<u8>>>>);
impl Keyring for MapKeyring {
    fn get(&self, account: &str) -> Result<Option<Vec<u8>>> {
        Ok(self.0.lock().expect("lock").get(account).cloned())
    }
    fn set(&self, account: &str, secret: &[u8]) -> Result<()> {
        self.0.lock().expect("lock").insert(account.to_string(), secret.to_vec());
        Ok(())
    }
    fn delete(&self, account: &str) -> Result<()> {
        self.0.lock().expect("lock").remove(account);
        Ok(())
    }
}

fn main() {
    let out = std::path::PathBuf::from(std::env::var("FIXTURE_OUT").expect("FIXTURE_OUT"));
    let data = out.join("app-data");
    std::fs::create_dir_all(&data).expect("mkdir");
    let kr = MapKeyring::default();
    let vault = CustodyVault::new(Box::new(kr.clone()), data.join("custody.enc"), 30);
    vault.ensure_auto_unlocked().expect("auto unlock");
    // The public Hardhat/Foundry test mnemonic: a well-known, unfunded-on-40204 test seed.
    let info = citrate_core_kit::wallet::import(
        &vault,
        "test test test test test test test test test test test junk",
    )
    .expect("import");
    // A backend-written connection record (connections.rs:806 at v0.5.1 writes the same slot name).
    let mut conn = br#"{"accessToken":"upgrade-fixture-not-a-token","refreshToken":null,"scope":"read","expiresAt":null,"linkedAt":1792300000}"#.to_vec();
    vault.put("connection-github", &mut conn).expect("put connection");
    let b64 = base64::engine::general_purpose::STANDARD;
    let map: BTreeMap<String, String> = kr
        .0
        .lock()
        .expect("lock")
        .iter()
        .map(|(k, v)| (k.clone(), b64.encode(v)))
        .collect();
    let meta = serde_json::json!({
        "generatedFrom": "v0.5.1 (b521d98e) kit::custody::CustodyVault + kit::wallet::import",
        "keyringService": "ai.citrate.core.custody",
        "keyring": map,
        "walletAddress": info.address,
        "walletPublicKeyHex": info.public_key_hex,
        "slots": ["wallet-entropy-0", "connection-github"],
    });
    std::fs::write(
        out.join("custody.keyring.json"),
        serde_json::to_vec_pretty(&meta).expect("json"),
    )
    .expect("write meta");

    // kit config.json as tauri-plugin-store 2.4.3 writes it (default_serialize = to_vec_pretty of
    // the {key: value} cache; kit/src/config.rs CONFIG_KEY = "app").
    let mut cfg = citrate_core_kit::config::AppConfig::default();
    cfg.autolock = 15;
    cfg.telemetry = true;
    cfg.sig_policy = "allow".into();
    let store = serde_json::json!({ "app": serde_json::to_value(&cfg).expect("cfg") });
    std::fs::write(
        data.join("config.json"),
        serde_json::to_vec_pretty(&store).expect("json"),
    )
    .expect("write config");
    println!("{}", info.address);
}
